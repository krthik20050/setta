# Project Setta — MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:phantom (recommended) to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Single-file Windows desktop pet that detects hand raise via webcam → plays greeting .wav + AI chat via OpenRouter.

**Architecture:** Single Python file `setta.py` with three concurrent loops: (1) webcam capture + MediaPipe hand detection in a background thread, (2) tkinter pet window (always-on-top, frameless, sprite animations), (3) OpenRouter streaming chat. State shared via threading.Event and a simple state enum.

**Tech Stack:** Python 3.13, tkinter, Pillow, MediaPipe, OpenCV, winsound (built-in), OpenRouter API.

**Spritesheet:** 1536×1872 WEBP → 9 rows × 8 cols, cell=192×208. Row mapping determined heuristically: row 0 (6 frames) = idle, row 4 (5 frames) = wave, row 7 (6 frames) = run/jump, row 8 (6 frames) = review, row 5 (8 frames) = coding/extra. Greeting trigger: hand raised (MediaPipe landmarks y-threshold).

class PetApp:
    def __init__(self):

        self.frame_idx = 0
        self.last_trigger = 0.0

    def _load_frames(self):
        frames = {}
        for row in range(ROWS):
            row_frames = []
            for col in range(COLS):
                x, y = col * CELL_W, row * CELL_H
                cell = self.spritesheet.crop((x, y, x + CELL_W, y + CELL_H))
                tk_img = ImageTk.PhotoImage(cell)
                row_frames.append(tk_img)
            frames[row] = row_frames
        return frames

- [ ] **Step 3: Implement _animate with idle → wave → idle state machine**

```python
    def _animate(self):
        row_map = {"idle": ROW_IDLE, "wave": ROW_WAVE,
                   "run": ROW_RUN, "review": ROW_REVIEW, "failed": ROW_FAILED}
        row = row_map.get(self.state, ROW_IDLE)
        frames_list = self.frames[row]

        if self.frame_idx >= len(frames_list):
            self.frame_idx = 0
            if self.state in ("wave", "run"):
                self.state = "idle"

        if frames_list:
            self.canvas.delete("sprite")
            self.canvas.create_image(CELL_W//2, CELL_H//2,
                                     image=frames_list[self.frame_idx],
                                     tags="sprite")
            self.frame_idx += 1

        # Check hand raise event
        if self.hand_raised.is_set():
            self.hand_raised.clear()
            now = time.time()
            if now - self.last_trigger > COOLDOWN_SECS:
                self.last_trigger = now
                self.state = "wave"
                self.frame_idx = 0
                self._play_greeting()

        self.root.after(150, self._animate)

    def _play_greeting(self):
        greeting = "greeting.wav"
        if os.path.exists(greeting):
            winsound.PlaySound(greeting, winsound.SND_ASYNC)
```

### Task 2: Webcam hand detection thread

- [ ] **Step 1: Add webcam thread to PetApp**

```python
    def start_webcam(self):
        def _capture():
            cap = cv2.VideoCapture(0)
            mp_hands = mp.solutions.hands
            hands = mp_hands.Hands(min_detection_confidence=0.7,
                                    min_tracking_confidence=0.5)

            while cap.isOpened():
                ret, frame = cap.read()
                if not ret:
                    continue
                h, w, _ = frame.shape
                rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                result = hands.process(rgb)

                if result.multi_hand_landmarks:
                    for hand_landmarks in result.multi_hand_landmarks:
                        wrist_y = hand_landmarks.landmark[0].y
                        if wrist_y < HAND_Y_THRESHOLD:
                            self.hand_raised.set()
                            break
                time.sleep(0.05)

            cap.release()

        t = threading.Thread(target=_capture, daemon=True)
        t.start()
```

### Task 3: OpenRouter chat

- [ ] **Step 1: Add OpenRouter call method**

```python
    def chat_with_ai(self, prompt):
        if not OPENROUTER_API_KEY:
            return "API key not configured."
        data = json.dumps({
            "model": OPENROUTER_MODEL,
            "messages": [{"role": "user", "content": prompt}],
            "stream": False
        }).encode()
        req = urllib.request.Request(OPENROUTER_URL, data=data,
            headers={
                "Authorization": f"Bearer {OPENROUTER_API_KEY}",
                "Content-Type": "application/json",
            })
        try:
            resp = urllib.request.urlopen(req, timeout=30)
            result = json.loads(resp.read())
            return result["choices"][0]["message"]["content"]
        except Exception as e:
            return f"Error: {e}"
```

    def run(self):
        self.canvas.bind("<Button-1>", self._drag_start)

    def _show_menu(self, event):
        from tkinter import simpledialog
        prompt = simpledialog.askstring("Setta Chat", "Ask Setta:")
        if prompt:
            self.state = "review"
            self.frame_idx = 0
            self.root.after(100, lambda: self._chat_response(prompt))
            self.show_speech_bubble("Thinking...")

    def _chat_response(self, prompt):
        reply = self.chat_with_ai(prompt)
        self.state = "idle"
        self.show_speech_bubble(reply)

    def show_speech_bubble(self, text, duration=5):
        self.canvas.delete("bubble")
        wrapped = "\n".join(text[i:i+40] for i in range(0, len(text), 40))
        self.canvas.create_text(CELL_W//2, 20, text=wrapped,
                                fill="white", font=("Segoe UI", 9),
                                width=180, tags="bubble")
        self.root.after(duration * 1000,
                        lambda: self.canvas.delete("bubble"))
